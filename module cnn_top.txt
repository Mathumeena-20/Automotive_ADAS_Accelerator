module cnn_top (
    input clk,
    input reset,
    output reg valid,
    output reg signed [15:0] final_out
);

// Input & Kernel
reg signed [7:0] p[0:8];
reg signed [7:0] k[0:8];

wire signed [15:0] conv_out;
wire signed [15:0] relu_out;

// Cycle counter
reg [3:0] count;

// Initialize
initial begin
    p[0]=1; p[1]=2; p[2]=3;
    p[3]=4; p[4]=5; p[5]=6;
    p[6]=7; p[7]=8; p[8]=9;

    k[0]=1; k[1]=0; k[2]=1;
    k[3]=0; k[4]=1; k[5]=0;
    k[6]=1; k[7]=0; k[8]=1;
end

// Convolution
conv3x3 conv_inst (
    .clk(clk),
    .reset(reset),
    .p0(p[0]), .p1(p[1]), .p2(p[2]),
    .p3(p[3]), .p4(p[4]), .p5(p[5]),
    .p6(p[6]), .p7(p[7]), .p8(p[8]),
    .k0(k[0]), .k1(k[1]), .k2(k[2]),
    .k3(k[3]), .k4(k[4]), .k5(k[5]),
    .k6(k[6]), .k7(k[7]), .k8(k[8]),
    .result(conv_out)
);

// ReLU
relu relu_inst (
    .in(conv_out),
    .out(relu_out)
);

// Control Logic (IMPORTANT)
always @(posedge clk or posedge reset) begin
    if (reset) begin
        count <= 0;
        valid <= 0;
        final_out <= 0;
    end else begin
        count <= count + 1;

        if (count == 3) begin
            final_out <= relu_out;
            valid <= 1;
        end
    end
end

endmodule
